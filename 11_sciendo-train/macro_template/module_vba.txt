Option Explicit

' Global updating flag to avoid recursive event triggers
Public IsUpdating As Boolean

' Display messages only when interacting with a human user (prevents hanging in automation)

' Language helper: checks if the workbook is in Bahasa Indonesia
Public Function IsIndonesian() As Boolean
    Dim dashWs As Worksheet
    On Error Resume Next
    Set dashWs = ThisWorkbook.Sheets("Dashboard")
    On Error GoTo 0
    If Not dashWs Is Nothing Then
        If InStr(1, CStr(dashWs.Cells(2, 2).Value), "Perubahan", vbTextCompare) > 0 Or _
           InStr(1, CStr(dashWs.Cells(7, 2).Value), "Kebijakan", vbTextCompare) > 0 Or _
           InStr(1, CStr(dashWs.Cells(12, 2).Value), "Panduan", vbTextCompare) > 0 Then
            IsIndonesian = True
            Exit Function
        End If
    End If
    If Not ActiveSheet Is Nothing Then
        If InStr(1, CStr(ActiveSheet.Cells(2, 2).Value), "Kembali", vbTextCompare) > 0 Or _
           InStr(1, CStr(ActiveSheet.Cells(3, 2).Value), "Panduan", vbTextCompare) > 0 Or _
           InStr(1, CStr(ActiveSheet.Cells(6, 2).Value), "Dari", vbTextCompare) > 0 Then
            IsIndonesian = True
            Exit Function
        End If
    End If
End Function

Public Function StatusIntervened() As String
    If IsIndonesian() Then StatusIntervened = "Diintervensi" Else StatusIntervened = "Intervened"
End Function

Public Function StatusBaseline() As String
    StatusBaseline = "Baseline"
End Function

Public Function StatusValid() As String
    StatusValid = "VALID"
End Function

Public Function StatusWarning() As String
    If IsIndonesian() Then StatusWarning = "PERINGATAN" Else StatusWarning = "WARNING"
End Function

Public Sub ShowMsg(ByVal msg As String, Optional ByVal style As VbMsgBoxStyle = vbInformation, Optional ByVal title As String = "SCIENDO Scenario Builder")
    If Application.Visible And Application.DisplayAlerts Then
        MsgBox msg, style, title
    End If
End Sub

' Safely retrieve baseline sheet respecting Excel 31-character limit
Public Function GetBaseSheet(ByVal ws As Worksheet) As Worksheet
    Dim baseName As String
    baseName = Left("_Base_" & ws.Name, 31)
    On Error Resume Next
    Set GetBaseSheet = ThisWorkbook.Sheets(baseName)
    On Error GoTo 0
End Function

' Safe double parsing supporting both dot and comma decimal notations
Public Function SafeCDbl(ByVal val As Variant, Optional ByVal defaultVal As Double = 0#) As Double
    On Error Resume Next
    If IsNumeric(val) Then
        SafeCDbl = CDbl(val)
    Else
        Dim s As String
        s = Replace(Trim(CStr(val)), ",", ".")
        SafeCDbl = Val(s)
    End If
    On Error GoTo 0
End Function

' Get initial class area in hectares for a given From class
Public Function GetInitialClassArea(ws As Worksheet, ByVal fromClassName As String, ByVal fromRow As Long, ByVal panelCol As Long, ByVal pStart As Long, ByVal pEnd As Long) As Double
    Dim r As Long
    If panelCol > 0 And pEnd >= pStart Then
        For r = pStart To pEnd
            If StrComp(Trim(CStr(ws.Cells(r, panelCol).Value)), Trim(fromClassName), vbTextCompare) = 0 Then
                Dim br As Double: br = SafeCDbl(ws.Cells(r, panelCol + 2).Value)
                Dim ba As Double: ba = SafeCDbl(ws.Cells(r, panelCol + 3).Value)
                If br > 0.000001 And ba > 0.0001 Then
                    GetInitialClassArea = Round(ba / br, 2)
                    Exit Function
                End If
            End If
        Next r
    End If
    
    ' Fallback: check baseline sheet if available
    Dim baseWs As Worksheet
    Set baseWs = GetBaseSheet(ws)
    If Not baseWs Is Nothing And panelCol > 0 Then
        Dim basePEnd As Long: basePEnd = baseWs.Cells(baseWs.Rows.Count, panelCol).End(xlUp).Row
        If basePEnd >= pStart Then
            For r = pStart To basePEnd
                If StrComp(Trim(CStr(baseWs.Cells(r, panelCol).Value)), Trim(fromClassName), vbTextCompare) = 0 Then
                    Dim bBr As Double: bBr = SafeCDbl(baseWs.Cells(r, panelCol + 2).Value)
                    Dim bBa As Double: bBa = SafeCDbl(baseWs.Cells(r, panelCol + 3).Value)
                    If bBr > 0.000001 And bBa > 0.0001 Then
                        GetInitialClassArea = Round(bBa / bBr, 2)
                        Exit Function
                    End If
                End If
            Next r
        End If
    End If
    
    ' Standard default if no other transition exists for this class
    GetInitialClassArea = 1000#
End Function

' Locate an existing row in the recapitulation panel corresponding to (mRow, mCol)
Public Function FindPanelRow(ws As Worksheet, ByVal mRow As Long, ByVal mCol As Long, ByVal panelCol As Long, ByVal pStart As Long, ByVal pEnd As Long) As Long
    FindPanelRow = 0
    If panelCol = 0 Or pEnd < pStart Then Exit Function
    Dim r As Long
    For r = pStart To pEnd
        If Val(ws.Cells(r, panelCol + 8).Value) = mRow And Val(ws.Cells(r, panelCol + 9).Value) = mCol Then
            FindPanelRow = r
            Exit Function
        End If
    Next r
End Function

' Cleanly delete a single row from the recapitulation panel without disturbing matrix cells
Public Sub DeletePanelRow(ws As Worksheet, ByVal delRow As Long, ByVal panelCol As Long)
    If delRow <= 0 Or panelCol <= 0 Then Exit Sub
    Dim pEnd As Long
    pEnd = ws.Cells(ws.Rows.Count, panelCol).End(xlUp).Row
    If delRow <= pEnd Then
        ws.Range(ws.Cells(delRow, panelCol), ws.Cells(delRow, panelCol + 9)).Delete Shift:=xlUp
    End If
End Sub

' Dynamically append a new 0-to-exist intervention row to the recapitulation panel
Public Function AppendPanelRow(ws As Worksheet, ByVal mRow As Long, ByVal mCol As Long, ByVal newVal As Double, _
                               ByVal matHeaderRow As Long, ByVal matLabelCol As Long, _
                               ByVal panelCol As Long, ByVal pStart As Long, ByRef pEnd As Long) As Long
    Dim newRow As Long
    If pEnd < pStart Then
        newRow = pStart
    Else
        newRow = pEnd + 1
    End If
    
    Dim fromName As String, toName As String
    fromName = CStr(ws.Cells(mRow, matLabelCol).Value)
    toName = CStr(ws.Cells(matHeaderRow, mCol).Value)
    
    ws.Cells(newRow, panelCol + 0).Value = fromName
    ws.Cells(newRow, panelCol + 1).Value = toName
    ws.Cells(newRow, panelCol + 2).Value = 0#
    ws.Cells(newRow, panelCol + 2).NumberFormat = "0.0000"
    ws.Cells(newRow, panelCol + 3).Value = 0#
    ws.Cells(newRow, panelCol + 3).NumberFormat = "#,##0.00"
    ws.Cells(newRow, panelCol + 4).NumberFormat = "0.0%"
    ws.Cells(newRow, panelCol + 4).Value = newVal
    ws.Cells(newRow, panelCol + 5).Value = newVal
    ws.Cells(newRow, panelCol + 5).NumberFormat = "0.0000"
    
    Dim initArea As Double
    initArea = GetInitialClassArea(ws, fromName, mRow, panelCol, pStart, pEnd)
    ws.Cells(newRow, panelCol + 6).FormulaR1C1 = "=ROUND(" & Replace(CStr(initArea), ",", ".") & "*RC[-1], 2)"
    ws.Cells(newRow, panelCol + 6).NumberFormat = "#,##0.00"
    
    ws.Cells(newRow, panelCol + 7).Value = StatusIntervened()
    ws.Cells(newRow, panelCol + 7).Font.Bold = True
    ws.Cells(newRow, panelCol + 7).Interior.Color = RGB(255, 235, 130)
    
    ws.Cells(newRow, panelCol + 8).Value = mRow
    ws.Cells(newRow, panelCol + 9).Value = mCol
    
    pEnd = newRow
    AppendPanelRow = newRow
End Function

' Trigger cell changes programmatically or via COM automation
Public Sub TriggerCellChange(ByVal wsName As String, ByVal rowNum As Long, ByVal colNum As Long, ByVal newVal As Variant)
    Dim ws As Worksheet
    On Error Resume Next
    Set ws = ThisWorkbook.Sheets(wsName)
    On Error GoTo 0
    If ws Is Nothing Then Exit Sub
    ws.Cells(rowNum, colNum).Value = newVal
    HandleSheetChange ws, ws.Cells(rowNum, colNum)
End Sub

' Handle cell changes on any sheet
Public Sub HandleSheetChange(ByVal ws As Worksheet, ByVal Target As Range)
    If IsUpdating Then Exit Sub
    If ws Is Nothing Then Exit Sub
    If ws.Name = "Dashboard" Or Left(ws.Name, 1) = "_" Then Exit Sub
    If Target.Cells.CountLarge > 1 Then Exit Sub ' Ignore bulk edits

    ' Direct geometry indexing (matrix starts at Row 4 or 6 in SCIENDO workbooks)
    Dim matHeaderRow As Long, matLabelCol As Long, matStartCol As Long
    matLabelCol = 2
    matStartCol = 3
    matHeaderRow = 6
    Dim rScanH As Long
    For rScanH = 4 To 8
        Dim hScanText As String
        hScanText = Trim(CStr(ws.Cells(rScanH, matLabelCol).Value))
        If InStr(1, hScanText, "From", vbTextCompare) > 0 Or InStr(1, hScanText, "Dari", vbTextCompare) > 0 Then
            matHeaderRow = rScanH
            Exit For
        End If
    Next rScanH
    
    ' Find Grand Total / Total Baris column
    Dim colScan As Long, matEndCol As Long, nClasses As Long
    colScan = matStartCol
    Do While ws.Cells(matHeaderRow, colScan).Value <> "" And _
             ws.Cells(matHeaderRow, colScan).Value <> "Grand Total" And _
             ws.Cells(matHeaderRow, colScan).Value <> "Total Baris"
        colScan = colScan + 1
        If colScan > 200 Then Exit Do
    Loop
    matEndCol = colScan - 1
    nClasses = matEndCol - matStartCol + 1
    If nClasses <= 0 Then Exit Sub
    
    Dim matStartRow As Long, matEndRow As Long, sumCol As Long
    matStartRow = matHeaderRow + 1
    matEndRow = matHeaderRow + nClasses
    sumCol = matEndCol + 1
    
    ' Locate Panel directly (placed 2 columns after Grand Total)
    ' Layout: [From Class | To Class | Baseline Rate | Baseline Area (ha) | % Adjustment | Scenario Rate | Scenario Area (ha) | Intervention Status | _mRow | _mCol]
    Dim panelCol As Long, panelRowStart As Long, panelRowEnd As Long
    panelCol = sumCol + 2
    Dim pScanTitle As String
    pScanTitle = CStr(ws.Cells(matHeaderRow, panelCol).Value)
    If InStr(1, pScanTitle, "From Class", vbTextCompare) > 0 Or _
       InStr(1, pScanTitle, "Kelas Asal", vbTextCompare) > 0 Then
        panelRowStart = matHeaderRow + 1
        panelRowEnd = ws.Cells(ws.Rows.Count, panelCol).End(xlUp).Row
    Else
        panelCol = 0
    End If
    
    Dim tRow As Long, tCol As Long
    tRow = Target.Row
    tCol = Target.Column
    
    Dim baseWs As Worksheet
    Set baseWs = GetBaseSheet(ws)
    
    ' CASE 1: Cell in Matrix changed
    If tRow >= matStartRow And tRow <= matEndRow And tCol >= matStartCol And tCol <= matEndCol Then
        If Not IsNumeric(Target.Value) Or SafeCDbl(Target.Value) < 0 Or SafeCDbl(Target.Value) > 1 Then
            If IsIndonesian() Then
                ShowMsg "Nilai tidak valid! Silakan masukkan probabilitas antara 0 dan 1.", vbExclamation, "Validasi Masukan"
            Else
                ShowMsg "Invalid value! Please enter a probability between 0 and 1.", vbExclamation, "Input Validation"
            End If
            IsUpdating = True
            Dim prevEvM1 As Boolean: prevEvM1 = Application.EnableEvents
            Application.EnableEvents = False
            If Not baseWs Is Nothing Then
                Target.Value = baseWs.Cells(tRow, tCol).Value
            Else
                Target.Value = 0#
            End If
            Application.EnableEvents = prevEvM1
            IsUpdating = False
            Exit Sub
        End If
        
        Dim newVal As Double
        newVal = SafeCDbl(Target.Value)
        
        ' Validate locked cells sum
        If Not ValidateLockedSum(ws, tRow, tCol, newVal, matStartCol, matEndCol, panelCol, panelRowStart, panelRowEnd) Then
            IsUpdating = True
            Dim prevEvM2 As Boolean: prevEvM2 = Application.EnableEvents
            Application.EnableEvents = False
            If Not baseWs Is Nothing Then
                Target.Value = baseWs.Cells(tRow, tCol).Value
            Else
                Target.Value = 0#
            End If
            Application.EnableEvents = prevEvM2
            IsUpdating = False
            Exit Sub
        End If
        
        Dim rIdx As Long, cIdx As Long, isDiagonal As Boolean
        rIdx = tRow - matHeaderRow
        cIdx = tCol - matLabelCol
        isDiagonal = (rIdx = cIdx)
        
        IsUpdating = True
        Dim prevEvents1 As Boolean
        prevEvents1 = Application.EnableEvents
        Application.EnableEvents = False
        
        If isDiagonal Then
            ' Diagonal persistence cell changed directly
            Target.Value = newVal
            Target.Interior.Color = RGB(255, 235, 130)
            RebalanceRow ws, tRow, tCol, newVal, matStartCol, matEndCol, panelCol, panelRowStart, panelRowEnd
            UpdateRowSumOnSheet ws, tRow, matStartCol, matEndCol, sumCol
        Else
            ' Off-diagonal transition cell changed
            Dim pRow As Long
            pRow = FindPanelRow(ws, tRow, tCol, panelCol, panelRowStart, panelRowEnd)
            
            If newVal > 0.000001 Then
                Target.Value = newVal
                Target.Interior.Color = RGB(255, 235, 130)
                
                If pRow > 0 Then
                    UpdatePanelFromMatrix ws, tRow, tCol, newVal, panelCol, panelRowStart, panelRowEnd
                Else
                    ' 0-to-exist dynamic append
                    AppendPanelRow ws, tRow, tCol, newVal, matHeaderRow, matLabelCol, panelCol, panelRowStart, panelRowEnd
                End If
                
                RebalanceRow ws, tRow, tCol, newVal, matStartCol, matEndCol, panelCol, panelRowStart, panelRowEnd
                UpdateRowSumOnSheet ws, tRow, matStartCol, matEndCol, sumCol
            Else
                ' newVal == 0
                Target.Value = 0#
                RestoreCellFormatting ws, tRow, tCol, matHeaderRow, matLabelCol
                
                If pRow > 0 Then
                    Dim bRateRow As Double
                    bRateRow = SafeCDbl(ws.Cells(pRow, panelCol + 2).Value)
                    If bRateRow <= 0.000001 Then
                        ' Dynamic 0-to-exist row returned to 0 -> cleanly delete from table
                        DeletePanelRow ws, pRow, panelCol
                        panelRowEnd = ws.Cells(ws.Rows.Count, panelCol).End(xlUp).Row
                    Else
                        ' Baseline transition set to 0 -> mark intervened with -100%
                        ws.Cells(pRow, panelCol + 4).Value = -1#
                        ws.Cells(pRow, panelCol + 5).Value = 0#
                        ws.Cells(pRow, panelCol + 6).FormulaR1C1 = "=ROUND(RC[-3]*(1+RC[-2]), 2)"
                        ws.Cells(pRow, panelCol + 7).Value = StatusIntervened()
                        ws.Cells(pRow, panelCol + 7).Font.Bold = True
                        ws.Cells(pRow, panelCol + 7).Interior.Color = RGB(255, 235, 130)
                        Target.Interior.Color = RGB(255, 235, 130)
                    End If
                End If
                
                RebalanceRow ws, tRow, tCol, 0#, matStartCol, matEndCol, panelCol, panelRowStart, panelRowEnd
                UpdateRowSumOnSheet ws, tRow, matStartCol, matEndCol, sumCol
            End If
        End If
        
        Application.EnableEvents = prevEvents1
        IsUpdating = False
        Exit Sub
    End If
    
    ' CASE 2: Cell in Panel changed
    If panelCol > 0 And tRow >= panelRowStart And tRow <= panelRowEnd Then
        Dim colOffset As Long
        colOffset = tCol - panelCol
        
        Dim mRow As Long, mCol As Long, baseRate As Double, scenRate As Double, pctAdj As Double
        mRow = Val(ws.Cells(tRow, panelCol + 8).Value)
        mCol = Val(ws.Cells(tRow, panelCol + 9).Value)
        baseRate = SafeCDbl(ws.Cells(tRow, panelCol + 2).Value)
        Dim isAppended As Boolean: isAppended = (baseRate <= 0.000001)
        
        If mRow >= matStartRow And mRow <= matEndRow And mCol >= matStartCol And mCol <= matEndCol Then
            ' Subcase 2A: % Adjustment column (offset 4)
            If colOffset = 4 Then
                If isAppended Then
                    If Not IsNumeric(Target.Value) Then
                        If IsIndonesian() Then
                            ShowMsg "Silakan masukkan persentase penyesuaian yang valid (contoh: 10% atau 0.10).", vbExclamation, "Persentase Tidak Valid"
                        Else
                            ShowMsg "Please enter a valid percentage adjustment (e.g. 10% or 0.10).", vbExclamation, "Invalid Percentage"
                        End If
                        IsUpdating = True
                        Dim prevEvPApE As Boolean: prevEvPApE = Application.EnableEvents
                        Application.EnableEvents = False
                        Target.Value = SafeCDbl(ws.Cells(mRow, mCol).Value)
                        Application.EnableEvents = prevEvPApE
                        IsUpdating = False
                        Exit Sub
                    End If
                    scenRate = SafeCDbl(Target.Value)
                    If Abs(scenRate) > 1# And Abs(scenRate) <= 100# Then
                        scenRate = scenRate / 100#
                    End If
                    If scenRate <= 0.000001 Then
                        ' Setting to 0 removes appended row cleanly
                        IsUpdating = True
                        Dim prevEvPAppDelA As Boolean: prevEvPAppDelA = Application.EnableEvents
                        Application.EnableEvents = False
                        ws.Cells(mRow, mCol).Value = 0#
                        RestoreCellFormatting ws, mRow, mCol, matHeaderRow, matLabelCol
                        DeletePanelRow ws, tRow, panelCol
                        panelRowEnd = ws.Cells(ws.Rows.Count, panelCol).End(xlUp).Row
                        RebalanceRow ws, mRow, mCol, 0#, matStartCol, matEndCol, panelCol, panelRowStart, panelRowEnd
                        UpdateRowSumOnSheet ws, mRow, matStartCol, matEndCol, sumCol
                        Application.EnableEvents = prevEvPAppDelA
                        IsUpdating = False
                        Exit Sub
                    Else
                        If scenRate > 1# Then scenRate = 1#
                        If scenRate < 0# Then scenRate = 0#
                        If Not ValidateLockedSum(ws, mRow, mCol, scenRate, matStartCol, matEndCol, panelCol, panelRowStart, panelRowEnd) Then
                            IsUpdating = True
                            Dim prevEvPValAppA As Boolean: prevEvPValAppA = Application.EnableEvents
                            Application.EnableEvents = False
                            Target.Value = SafeCDbl(ws.Cells(mRow, mCol).Value)
                            Application.EnableEvents = prevEvPValAppA
                            IsUpdating = False
                            Exit Sub
                        End If
                        IsUpdating = True
                        Dim prevEvPAppOkA As Boolean: prevEvPAppOkA = Application.EnableEvents
                        Application.EnableEvents = False
                        Target.NumberFormat = "0.0%"
                        Target.Value = scenRate
                        ws.Cells(tRow, panelCol + 5).Value = scenRate
                        Dim initA1A As Double
                        initA1A = GetInitialClassArea(ws, CStr(ws.Cells(tRow, panelCol).Value), mRow, panelCol, panelRowStart, panelRowEnd)
                        ws.Cells(tRow, panelCol + 6).FormulaR1C1 = "=ROUND(" & Replace(CStr(initA1A), ",", ".") & "*RC[-1], 2)"
                        ws.Cells(tRow, panelCol + 7).Value = StatusIntervened()
                        ws.Cells(tRow, panelCol + 7).Font.Bold = True
                        ws.Cells(tRow, panelCol + 7).Interior.Color = RGB(255, 235, 130)
                        ws.Cells(mRow, mCol).Value = scenRate
                        ws.Cells(mRow, mCol).Interior.Color = RGB(255, 235, 130)
                        RebalanceRow ws, mRow, mCol, scenRate, matStartCol, matEndCol, panelCol, panelRowStart, panelRowEnd
                        UpdateRowSumOnSheet ws, mRow, matStartCol, matEndCol, sumCol
                        Application.EnableEvents = prevEvPAppOkA
                        IsUpdating = False
                        Exit Sub
                    End If
                End If
                
                If Not IsNumeric(Target.Value) Then
                    If IsIndonesian() Then
                        ShowMsg "Silakan masukkan persentase penyesuaian yang valid (contoh: -0.2 untuk -20%).", vbExclamation, "Persentase Tidak Valid"
                    Else
                        ShowMsg "Please enter a valid percentage adjustment (e.g. -0.2 for -20%).", vbExclamation, "Invalid Percentage"
                    End If
                    IsUpdating = True
                    Dim prevEvP1 As Boolean: prevEvP1 = Application.EnableEvents
                    Application.EnableEvents = False
                    Dim curScen1 As Double: curScen1 = SafeCDbl(ws.Cells(tRow, panelCol + 5).Value)
                    If baseRate > 0.000001 Then
                        Target.Value = (curScen1 - baseRate) / baseRate
                    Else
                        Target.Value = 0#
                    End If
                    Application.EnableEvents = prevEvP1
                    IsUpdating = False
                    Exit Sub
                End If
                pctAdj = SafeCDbl(Target.Value)
                If Abs(pctAdj) > 1# And Abs(pctAdj) <= 100# Then
                    pctAdj = pctAdj / 100#
                End If
                scenRate = baseRate * (1# + pctAdj)
                If scenRate < 0 Then scenRate = 0
                If scenRate > 1 Then scenRate = 1
                If baseRate > 0.000001 Then
                    pctAdj = (scenRate - baseRate) / baseRate
                Else
                    pctAdj = 0#
                End If
                
                If Not ValidateLockedSum(ws, mRow, mCol, scenRate, matStartCol, matEndCol, panelCol, panelRowStart, panelRowEnd) Then
                    IsUpdating = True
                    Dim prevEvP2 As Boolean: prevEvP2 = Application.EnableEvents
                    Application.EnableEvents = False
                    Dim curScen2 As Double: curScen2 = SafeCDbl(ws.Cells(tRow, panelCol + 5).Value)
                    If baseRate > 0.000001 Then
                        Target.Value = (curScen2 - baseRate) / baseRate
                    Else
                        Target.Value = 0#
                    End If
                    Application.EnableEvents = prevEvP2
                    IsUpdating = False
                    Exit Sub
                End If
                
                IsUpdating = True
                Dim prevEvents2A As Boolean
                prevEvents2A = Application.EnableEvents
                Application.EnableEvents = False
                
                Target.Value = pctAdj
                ws.Cells(tRow, panelCol + 5).Value = scenRate
                ws.Cells(tRow, panelCol + 6).FormulaR1C1 = "=ROUND(RC[-3]*(1+RC[-2]), 2)"
                ws.Cells(tRow, panelCol + 7).Value = StatusIntervened()
                ws.Cells(tRow, panelCol + 7).Font.Bold = True
                ws.Cells(tRow, panelCol + 7).Interior.Color = RGB(255, 235, 130)
                ws.Cells(mRow, mCol).Value = scenRate
                ws.Cells(mRow, mCol).Interior.Color = RGB(255, 235, 130)
                
                RebalanceRow ws, mRow, mCol, scenRate, matStartCol, matEndCol, panelCol, panelRowStart, panelRowEnd
                UpdateRowSumOnSheet ws, mRow, matStartCol, matEndCol, sumCol
                
                Application.EnableEvents = prevEvents2A
                IsUpdating = False
                Exit Sub
            End If
            
            ' Subcase 2B: Scenario Rate column (offset 5)
            If colOffset = 5 Then
                If Not IsNumeric(Target.Value) Or SafeCDbl(Target.Value) < 0 Or SafeCDbl(Target.Value) > 1 Then
                    If IsIndonesian() Then
                        ShowMsg "Silakan masukkan laju skenario yang valid antara 0 dan 1.", vbExclamation, "Laju Skenario Tidak Valid"
                    Else
                        ShowMsg "Please enter a valid scenario rate between 0 and 1.", vbExclamation, "Invalid Scenario Rate"
                    End If
                    IsUpdating = True
                    Dim prevEvP3 As Boolean: prevEvP3 = Application.EnableEvents
                    Application.EnableEvents = False
                    Target.Value = SafeCDbl(ws.Cells(mRow, mCol).Value)
                    Application.EnableEvents = prevEvP3
                    IsUpdating = False
                    Exit Sub
                End If
                scenRate = SafeCDbl(Target.Value)
                
                If isAppended Then
                    If scenRate <= 0.000001 Then
                        ' Revert to 0 -> delete appended row
                        IsUpdating = True
                        Dim prevEvPAppDel As Boolean: prevEvPAppDel = Application.EnableEvents
                        Application.EnableEvents = False
                        ws.Cells(mRow, mCol).Value = 0#
                        RestoreCellFormatting ws, mRow, mCol, matHeaderRow, matLabelCol
                        DeletePanelRow ws, tRow, panelCol
                        panelRowEnd = ws.Cells(ws.Rows.Count, panelCol).End(xlUp).Row
                        RebalanceRow ws, mRow, mCol, 0#, matStartCol, matEndCol, panelCol, panelRowStart, panelRowEnd
                        UpdateRowSumOnSheet ws, mRow, matStartCol, matEndCol, sumCol
                        Application.EnableEvents = prevEvPAppDel
                        IsUpdating = False
                        Exit Sub
                    Else
                        If Not ValidateLockedSum(ws, mRow, mCol, scenRate, matStartCol, matEndCol, panelCol, panelRowStart, panelRowEnd) Then
                            IsUpdating = True
                            Dim prevEvPValApp As Boolean: prevEvPValApp = Application.EnableEvents
                            Application.EnableEvents = False
                            Target.Value = SafeCDbl(ws.Cells(mRow, mCol).Value)
                            Application.EnableEvents = prevEvPValApp
                            IsUpdating = False
                            Exit Sub
                        End If
                        IsUpdating = True
                        Dim prevEvPAppOk As Boolean: prevEvPAppOk = Application.EnableEvents
                        Application.EnableEvents = False
                        Target.Value = scenRate
                        ws.Cells(tRow, panelCol + 4).NumberFormat = "0.0%"
                        ws.Cells(tRow, panelCol + 4).Value = scenRate
                        Dim initA1 As Double
                        initA1 = GetInitialClassArea(ws, CStr(ws.Cells(tRow, panelCol).Value), mRow, panelCol, panelRowStart, panelRowEnd)
                        ws.Cells(tRow, panelCol + 6).FormulaR1C1 = "=ROUND(" & Replace(CStr(initA1), ",", ".") & "*RC[-1], 2)"
                        ws.Cells(tRow, panelCol + 7).Value = StatusIntervened()
                        ws.Cells(tRow, panelCol + 7).Font.Bold = True
                        ws.Cells(tRow, panelCol + 7).Interior.Color = RGB(255, 235, 130)
                        ws.Cells(mRow, mCol).Value = scenRate
                        ws.Cells(mRow, mCol).Interior.Color = RGB(255, 235, 130)
                        RebalanceRow ws, mRow, mCol, scenRate, matStartCol, matEndCol, panelCol, panelRowStart, panelRowEnd
                        UpdateRowSumOnSheet ws, mRow, matStartCol, matEndCol, sumCol
                        Application.EnableEvents = prevEvPAppOk
                        IsUpdating = False
                        Exit Sub
                    End If
                Else
                    ' Standard baseline row
                    If baseRate > 0.000001 Then
                        pctAdj = (scenRate - baseRate) / baseRate
                    Else
                        pctAdj = 0#
                    End If
                    
                    If Not ValidateLockedSum(ws, mRow, mCol, scenRate, matStartCol, matEndCol, panelCol, panelRowStart, panelRowEnd) Then
                        IsUpdating = True
                        Dim prevEvP4 As Boolean: prevEvP4 = Application.EnableEvents
                        Application.EnableEvents = False
                        Target.Value = SafeCDbl(ws.Cells(mRow, mCol).Value)
                        Application.EnableEvents = prevEvP4
                        IsUpdating = False
                        Exit Sub
                    End If
                    
                    IsUpdating = True
                    Dim prevEvents2B As Boolean
                    prevEvents2B = Application.EnableEvents
                    Application.EnableEvents = False
                    
                    ws.Cells(tRow, panelCol + 4).Value = pctAdj
                    Target.Value = scenRate
                    ws.Cells(tRow, panelCol + 6).FormulaR1C1 = "=ROUND(RC[-3]*(1+RC[-2]), 2)"
                    ws.Cells(tRow, panelCol + 7).Value = StatusIntervened()
                    ws.Cells(tRow, panelCol + 7).Font.Bold = True
                    ws.Cells(tRow, panelCol + 7).Interior.Color = RGB(255, 235, 130)
                    ws.Cells(mRow, mCol).Value = scenRate
                    ws.Cells(mRow, mCol).Interior.Color = RGB(255, 235, 130)
                    
                    RebalanceRow ws, mRow, mCol, scenRate, matStartCol, matEndCol, panelCol, panelRowStart, panelRowEnd
                    UpdateRowSumOnSheet ws, mRow, matStartCol, matEndCol, sumCol
                    
                    Application.EnableEvents = prevEvents2B
                    IsUpdating = False
                    Exit Sub
                End If
            End If
            
            ' Subcase 2C: Scenario Area column (offset 6)
            If colOffset = 6 Then
                If Not IsNumeric(Target.Value) Or SafeCDbl(Target.Value) < 0 Then
                    If IsIndonesian() Then
                        ShowMsg "Silakan masukkan luas skenario positif yang valid dalam hektar.", vbExclamation, "Luas Skenario Tidak Valid"
                    Else
                        ShowMsg "Please enter a valid positive area in hectares.", vbExclamation, "Invalid Scenario Area"
                    End If
                    IsUpdating = True
                    Dim prevEvP5 As Boolean: prevEvP5 = Application.EnableEvents
                    Application.EnableEvents = False
                    If isAppended Then
                        Dim initAErr As Double
                        initAErr = GetInitialClassArea(ws, CStr(ws.Cells(tRow, panelCol).Value), mRow, panelCol, panelRowStart, panelRowEnd)
                        Target.FormulaR1C1 = "=ROUND(" & Replace(CStr(initAErr), ",", ".") & "*RC[-1], 2)"
                    Else
                        Target.FormulaR1C1 = "=ROUND(RC[-3]*(1+RC[-2]), 2)"
                    End If
                    Application.EnableEvents = prevEvP5
                    IsUpdating = False
                    Exit Sub
                End If
                
                Dim scenAreaInput As Double
                scenAreaInput = SafeCDbl(Target.Value)
                
                If isAppended Then
                    If scenAreaInput <= 0.0001 Then
                        ' Setting 0 area deletes appended row
                        IsUpdating = True
                        Dim prevEvPAreaDel As Boolean: prevEvPAreaDel = Application.EnableEvents
                        Application.EnableEvents = False
                        ws.Cells(mRow, mCol).Value = 0#
                        RestoreCellFormatting ws, mRow, mCol, matHeaderRow, matLabelCol
                        DeletePanelRow ws, tRow, panelCol
                        panelRowEnd = ws.Cells(ws.Rows.Count, panelCol).End(xlUp).Row
                        RebalanceRow ws, mRow, mCol, 0#, matStartCol, matEndCol, panelCol, panelRowStart, panelRowEnd
                        UpdateRowSumOnSheet ws, mRow, matStartCol, matEndCol, sumCol
                        Application.EnableEvents = prevEvPAreaDel
                        IsUpdating = False
                        Exit Sub
                    Else
                        Dim initA2 As Double
                        initA2 = GetInitialClassArea(ws, CStr(ws.Cells(tRow, panelCol).Value), mRow, panelCol, panelRowStart, panelRowEnd)
                        If initA2 > 0 Then scenRate = scenAreaInput / initA2 Else scenRate = 0#
                        If scenRate > 1# Then scenRate = 1#
                        
                        If Not ValidateLockedSum(ws, mRow, mCol, scenRate, matStartCol, matEndCol, panelCol, panelRowStart, panelRowEnd) Then
                            IsUpdating = True
                            Dim prevEvPValArea As Boolean: prevEvPValArea = Application.EnableEvents
                            Application.EnableEvents = False
                            Target.FormulaR1C1 = "=ROUND(" & Replace(CStr(initA2), ",", ".") & "*RC[-1], 2)"
                            Application.EnableEvents = prevEvPValArea
                            IsUpdating = False
                            Exit Sub
                        End If
                        
                        IsUpdating = True
                        Dim prevEvPAreaOk As Boolean: prevEvPAreaOk = Application.EnableEvents
                        Application.EnableEvents = False
                        ws.Cells(tRow, panelCol + 4).NumberFormat = "0.0%"
                        ws.Cells(tRow, panelCol + 4).Value = scenRate
                        ws.Cells(tRow, panelCol + 5).Value = scenRate
                        Target.FormulaR1C1 = "=ROUND(" & Replace(CStr(initA2), ",", ".") & "*RC[-1], 2)"
                        ws.Cells(tRow, panelCol + 7).Value = StatusIntervened()
                        ws.Cells(tRow, panelCol + 7).Font.Bold = True
                        ws.Cells(tRow, panelCol + 7).Interior.Color = RGB(255, 235, 130)
                        ws.Cells(mRow, mCol).Value = scenRate
                        ws.Cells(mRow, mCol).Interior.Color = RGB(255, 235, 130)
                        RebalanceRow ws, mRow, mCol, scenRate, matStartCol, matEndCol, panelCol, panelRowStart, panelRowEnd
                        UpdateRowSumOnSheet ws, mRow, matStartCol, matEndCol, sumCol
                        Application.EnableEvents = prevEvPAreaOk
                        IsUpdating = False
                        Exit Sub
                    End If
                Else
                    ' Standard baseline row
                    Dim baseAreaVal As Double
                    baseAreaVal = SafeCDbl(ws.Cells(tRow, panelCol + 3).Value)
                    If baseAreaVal > 0.0001 Then
                        pctAdj = (scenAreaInput - baseAreaVal) / baseAreaVal
                    Else
                        pctAdj = 0#
                    End If
                    scenRate = baseRate * (1# + pctAdj)
                    If scenRate < 0# Then scenRate = 0#
                    If scenRate > 1# Then scenRate = 1#
                    
                    If Not ValidateLockedSum(ws, mRow, mCol, scenRate, matStartCol, matEndCol, panelCol, panelRowStart, panelRowEnd) Then
                        IsUpdating = True
                        Dim prevEvP6 As Boolean: prevEvP6 = Application.EnableEvents
                        Application.EnableEvents = False
                        Target.FormulaR1C1 = "=ROUND(RC[-3]*(1+RC[-2]), 2)"
                        Application.EnableEvents = prevEvP6
                        IsUpdating = False
                        Exit Sub
                    End If
                    
                    IsUpdating = True
                    Dim prevEvents2C As Boolean
                    prevEvents2C = Application.EnableEvents
                    Application.EnableEvents = False
                    
                    ws.Cells(tRow, panelCol + 4).Value = pctAdj
                    ws.Cells(tRow, panelCol + 5).Value = scenRate
                    Target.FormulaR1C1 = "=ROUND(RC[-3]*(1+RC[-2]), 2)"
                    ws.Cells(tRow, panelCol + 7).Value = StatusIntervened()
                    ws.Cells(tRow, panelCol + 7).Font.Bold = True
                    ws.Cells(tRow, panelCol + 7).Interior.Color = RGB(255, 235, 130)
                    ws.Cells(mRow, mCol).Value = scenRate
                    ws.Cells(mRow, mCol).Interior.Color = RGB(255, 235, 130)
                    
                    RebalanceRow ws, mRow, mCol, scenRate, matStartCol, matEndCol, panelCol, panelRowStart, panelRowEnd
                    UpdateRowSumOnSheet ws, mRow, matStartCol, matEndCol, sumCol
                    
                    Application.EnableEvents = prevEvents2C
                    IsUpdating = False
                    Exit Sub
                End If
            End If
            
            ' Subcase 2D: Intervention Status column (offset 7)
            If colOffset = 7 Then
                Dim statusVal As String
                statusVal = Trim(UCase(Target.Value))
                
                If isAppended Then
                    If statusVal = "BASELINE" Or statusVal = "RESET" Or statusVal = "UNLOCKED" Or statusVal = "UNLOCK" Or statusVal = "NO" Then
                        IsUpdating = True
                        Dim prevEvPAppStat As Boolean: prevEvPAppStat = Application.EnableEvents
                        Application.EnableEvents = False
                        ws.Cells(mRow, mCol).Value = 0#
                        RestoreCellFormatting ws, mRow, mCol, matHeaderRow, matLabelCol
                        DeletePanelRow ws, tRow, panelCol
                        panelRowEnd = ws.Cells(ws.Rows.Count, panelCol).End(xlUp).Row
                        RebalanceRow ws, mRow, mCol, 0#, matStartCol, matEndCol, panelCol, panelRowStart, panelRowEnd
                        UpdateRowSumOnSheet ws, mRow, matStartCol, matEndCol, sumCol
                        Application.EnableEvents = prevEvPAppStat
                        IsUpdating = False
                        Exit Sub
                    Else
                        IsUpdating = True
                        Dim prevEvPAppStat2 As Boolean: prevEvPAppStat2 = Application.EnableEvents
                        Application.EnableEvents = False
                        Target.Value = StatusIntervened()
                        Target.Font.Bold = True
                        Target.Interior.Color = RGB(255, 235, 130)
                        ws.Cells(mRow, mCol).Interior.Color = RGB(255, 235, 130)
                        Application.EnableEvents = prevEvPAppStat2
                        IsUpdating = False
                        Exit Sub
                    End If
                Else
                    IsUpdating = True
                    Dim prevEvents2D As Boolean
                    prevEvents2D = Application.EnableEvents
                    Application.EnableEvents = False
                    
                    If statusVal = "BASELINE" Or statusVal = "UNLOCKED" Or statusVal = "UNLOCK" Or statusVal = "NO" Or statusVal = "RESET" Then
                        Target.Value = "Baseline"
                        Target.Font.Bold = False
                        Target.Interior.ColorIndex = xlNone
                        ws.Cells(tRow, panelCol + 4).Value = 0#
                        ws.Cells(tRow, panelCol + 5).Value = baseRate
                        ws.Cells(tRow, panelCol + 6).FormulaR1C1 = "=ROUND(RC[-3]*(1+RC[-2]), 2)"
                        ws.Cells(mRow, mCol).Value = baseRate
                        RestoreCellFormatting ws, mRow, mCol, matHeaderRow, matLabelCol
                        RebalanceRow ws, mRow, mCol, baseRate, matStartCol, matEndCol, panelCol, panelRowStart, panelRowEnd
                        UpdateRowSumOnSheet ws, mRow, matStartCol, matEndCol, sumCol
                    Else
                        Target.Value = StatusIntervened()
                        Target.Font.Bold = True
                        Target.Interior.Color = RGB(255, 235, 130)
                        ws.Cells(mRow, mCol).Interior.Color = RGB(255, 235, 130)
                    End If
                    
                    Application.EnableEvents = prevEvents2D
                    IsUpdating = False
                    Exit Sub
                End If
            End If
        End If
    End If
End Sub

' Validate that locked cells in the row do not sum to > 1
Function ValidateLockedSum(ws As Worksheet, rowNum As Long, changedCol As Long, newVal As Double, _
                          matStartCol As Long, matEndCol As Long, _
                          panelCol As Long, pStart As Long, pEnd As Long) As Boolean
    Dim c As Long, lockedSum As Double
    lockedSum = 0
    For c = matStartCol To matEndCol
        If c = changedCol Then
            lockedSum = lockedSum + newVal
        ElseIf IsCellLockedOnSheet(ws, rowNum, c, panelCol, pStart, pEnd) Then
            lockedSum = lockedSum + SafeCDbl(ws.Cells(rowNum, c).Value)
        End If
    Next c
    
    If lockedSum > 1.00001 Then
        If IsIndonesian() Then
            ShowMsg "Tidak dapat mengubah sel: Total sel terintervensi pada baris ini akan menjadi " & _
                   Format(lockedSum, "0.0000") & " yang melebihi 1.000." & vbCrLf & _
                   "Silakan masukkan nilai lebih kecil atau atur ulang sel terintervensi lainnya terlebih dahulu.", vbExclamation, "Batasan Total Baris"
        Else
            ShowMsg "Cannot modify cell: The sum of intervened cells in this row would be " & _
                   Format(lockedSum, "0.0000") & " which exceeds 1.000." & vbCrLf & _
                   "Please enter a smaller value or reset other intervened cells first.", vbExclamation, "Row Sum Constraint"
        End If
        ValidateLockedSum = False
    Else
        ValidateLockedSum = True
    End If
End Function

' Check if a cell is currently marked intervened / locked
Function IsCellLockedOnSheet(ws As Worksheet, rowNum As Long, colNum As Long, _
                            panelCol As Long, pStart As Long, pEnd As Long) As Boolean
    If panelCol > 0 And pEnd >= pStart Then
        Dim r As Long
        For r = pStart To pEnd
            If Val(ws.Cells(r, panelCol + 8).Value) = rowNum And Val(ws.Cells(r, panelCol + 9).Value) = colNum Then
                Dim st As String: st = UCase(Trim(ws.Cells(r, panelCol + 7).Value))
                If st = "INTERVENED" Or st = "DIINTERVENSI" Or st = "LOCKED" Then
                    IsCellLockedOnSheet = True
                    Exit Function
                Else
                    IsCellLockedOnSheet = False
                    Exit Function
                End If
            End If
        Next r
    End If
    
    ' Check cell background color (amber yellow indicates user-modified/locked)
    If ws.Cells(rowNum, colNum).Interior.Color = RGB(255, 235, 130) Then
        IsCellLockedOnSheet = True
    Else
        IsCellLockedOnSheet = False
    End If
End Function

' Rebalance unlocked cells in a row to maintain sum = 1.0
Sub RebalanceRow(ws As Worksheet, rowNum As Long, changedCol As Long, newVal As Double, _
                 matStartCol As Long, matEndCol As Long, _
                 panelCol As Long, pStart As Long, pEnd As Long)
    Dim c As Long
    Dim lockedSum As Double, totalUnlockedOriginal As Double
    Dim unlockedCols() As Long, nUnlocked As Long
    ReDim unlockedCols(1 To (matEndCol - matStartCol + 1))
    nUnlocked = 0
    lockedSum = 0
    totalUnlockedOriginal = 0
    
    ws.Cells(rowNum, changedCol).Value = newVal
    
    For c = matStartCol To matEndCol
        If c = changedCol Then
            lockedSum = lockedSum + newVal
        ElseIf IsCellLockedOnSheet(ws, rowNum, c, panelCol, pStart, pEnd) Then
            lockedSum = lockedSum + SafeCDbl(ws.Cells(rowNum, c).Value)
        End If
    Next c
    
    For c = matStartCol To matEndCol
        If c <> changedCol And Not IsCellLockedOnSheet(ws, rowNum, c, panelCol, pStart, pEnd) Then
            nUnlocked = nUnlocked + 1
            unlockedCols(nUnlocked) = c
            totalUnlockedOriginal = totalUnlockedOriginal + SafeCDbl(ws.Cells(rowNum, c).Value)
        End If
    Next c
    
    Dim remaining As Double
    remaining = 1# - lockedSum
    If remaining < 0 Then remaining = 0
    
    If nUnlocked = 0 Then Exit Sub
    
    Dim i As Long, targetCol As Long, lastCol As Long, lastVal As Double
    If remaining <= 0.0000001 Then
        For c = 1 To nUnlocked
            ws.Cells(rowNum, unlockedCols(c)).Value = 0
            RestoreCellFormatting ws, rowNum, unlockedCols(c)
            SyncMatrixToPanel ws, rowNum, unlockedCols(c), 0, panelCol, pStart, pEnd
        Next c
    ElseIf totalUnlockedOriginal > 0.000001 Then
        Dim runSum As Double: runSum = 0
        For i = 1 To nUnlocked - 1
            targetCol = unlockedCols(i)
            Dim propVal As Double
            propVal = (SafeCDbl(ws.Cells(rowNum, targetCol).Value) / totalUnlockedOriginal) * remaining
            If propVal < 0 Then propVal = 0
            ws.Cells(rowNum, targetCol).Value = propVal
            runSum = runSum + propVal
            RestoreCellFormatting ws, rowNum, targetCol
            SyncMatrixToPanel ws, rowNum, targetCol, propVal, panelCol, pStart, pEnd
        Next i
        lastCol = unlockedCols(nUnlocked)
        lastVal = remaining - runSum
        If lastVal < 0 Then lastVal = 0
        ws.Cells(rowNum, lastCol).Value = lastVal
        RestoreCellFormatting ws, rowNum, lastCol
        SyncMatrixToPanel ws, rowNum, lastCol, lastVal, panelCol, pStart, pEnd
    Else
        Dim eqVal As Double: eqVal = remaining / nUnlocked
        Dim runSum2 As Double: runSum2 = 0
        For i = 1 To nUnlocked - 1
            targetCol = unlockedCols(i)
            ws.Cells(rowNum, targetCol).Value = eqVal
            runSum2 = runSum2 + eqVal
            RestoreCellFormatting ws, rowNum, targetCol
            SyncMatrixToPanel ws, rowNum, targetCol, eqVal, panelCol, pStart, pEnd
        Next i
        lastCol = unlockedCols(nUnlocked)
        lastVal = remaining - runSum2
        If lastVal < 0 Then lastVal = 0
        ws.Cells(rowNum, lastCol).Value = lastVal
        RestoreCellFormatting ws, rowNum, lastCol
        SyncMatrixToPanel ws, rowNum, lastCol, lastVal, panelCol, pStart, pEnd
    End If
End Sub

' Update row sum display cell
Sub UpdateRowSumOnSheet(ws As Worksheet, rowNum As Long, matStartCol As Long, matEndCol As Long, sumCol As Long)
    Dim c As Long, rowSum As Double: rowSum = 0
    For c = matStartCol To matEndCol
        If IsNumeric(ws.Cells(rowNum, c).Value) Then
            rowSum = rowSum + SafeCDbl(ws.Cells(rowNum, c).Value)
        End If
    Next c
    If Abs(rowSum - 1#) < 0.000001 Then rowSum = 1#
    ws.Cells(rowNum, sumCol).Value = rowSum
End Sub

' Sync matrix value to panel row
Sub SyncMatrixToPanel(ws As Worksheet, rowNum As Long, colNum As Long, newVal As Double, _
                      panelCol As Long, pStart As Long, pEnd As Long)
    If panelCol = 0 Or pEnd < pStart Then Exit Sub
    Dim r As Long
    For r = pStart To pEnd
        If Val(ws.Cells(r, panelCol + 8).Value) = rowNum And Val(ws.Cells(r, panelCol + 9).Value) = colNum Then
            ws.Cells(r, panelCol + 5).Value = newVal
            Dim baseRate As Double
            baseRate = SafeCDbl(ws.Cells(r, panelCol + 2).Value)
            If baseRate > 0.000001 Then
                ws.Cells(r, panelCol + 4).Value = (newVal - baseRate) / baseRate
                ws.Cells(r, panelCol + 6).FormulaR1C1 = "=ROUND(RC[-3]*(1+RC[-2]), 2)"
            Else
                ws.Cells(r, panelCol + 4).NumberFormat = "0.0%"
                ws.Cells(r, panelCol + 4).Value = newVal
                Dim initA As Double
                initA = GetInitialClassArea(ws, CStr(ws.Cells(r, panelCol).Value), rowNum, panelCol, pStart, pEnd)
                ws.Cells(r, panelCol + 6).FormulaR1C1 = "=ROUND(" & Replace(CStr(initA), ",", ".") & "*RC[-1], 2)"
            End If
            Exit Sub
        End If
    Next r
End Sub

' Update panel row from modified matrix cell and mark intervened
Sub UpdatePanelFromMatrix(ws As Worksheet, rowNum As Long, colNum As Long, newVal As Double, _
                          panelCol As Long, pStart As Long, pEnd As Long)
    If panelCol = 0 Or pEnd < pStart Then Exit Sub
    Dim r As Long
    For r = pStart To pEnd
        If Val(ws.Cells(r, panelCol + 8).Value) = rowNum And Val(ws.Cells(r, panelCol + 9).Value) = colNum Then
            ws.Cells(r, panelCol + 5).Value = newVal
            Dim baseRate As Double
            baseRate = SafeCDbl(ws.Cells(r, panelCol + 2).Value)
            If baseRate > 0.000001 Then
                ws.Cells(r, panelCol + 4).Value = (newVal - baseRate) / baseRate
                ws.Cells(r, panelCol + 6).FormulaR1C1 = "=ROUND(RC[-3]*(1+RC[-2]), 2)"
            Else
                ws.Cells(r, panelCol + 4).NumberFormat = "0.0%"
                ws.Cells(r, panelCol + 4).Value = newVal
                Dim initA As Double
                initA = GetInitialClassArea(ws, CStr(ws.Cells(r, panelCol).Value), rowNum, panelCol, pStart, pEnd)
                ws.Cells(r, panelCol + 6).FormulaR1C1 = "=ROUND(" & Replace(CStr(initA), ",", ".") & "*RC[-1], 2)"
            End If
            ws.Cells(r, panelCol + 7).Value = StatusIntervened()
            ws.Cells(r, panelCol + 7).Font.Bold = True
            ws.Cells(r, panelCol + 7).Interior.Color = RGB(255, 235, 130)
            Exit Sub
        End If
    Next r
End Sub

' Restore default cell background formatting
Sub RestoreCellFormatting(ws As Worksheet, rowNum As Long, colNum As Long, Optional matHeaderRow As Long = 0, Optional matLabelCol As Long = 0)
    If matHeaderRow = 0 Then matHeaderRow = 4
    If matLabelCol = 0 Then matLabelCol = 2
    
    Dim rIdx As Long, cIdx As Long
    rIdx = rowNum - matHeaderRow
    cIdx = colNum - matLabelCol
    
    If rIdx = cIdx Then
        ws.Cells(rowNum, colNum).Interior.Color = RGB(254, 249, 195) ' Soft yellow (#FEF9C3) for diagonal persistence
        ws.Cells(rowNum, colNum).Font.Color = RGB(0, 0, 0)
    Else
        Dim cellVal As Double: cellVal = 0#
        If IsNumeric(ws.Cells(rowNum, colNum).Value) Then
            cellVal = SafeCDbl(ws.Cells(rowNum, colNum).Value)
        End If
        If cellVal > 0.000001 Then
            ws.Cells(rowNum, colNum).Interior.Color = RGB(219, 234, 254) ' Subtle blue (#DBEAFE) for active transition
            ws.Cells(rowNum, colNum).Font.Color = RGB(0, 0, 0)
        Else
            ws.Cells(rowNum, colNum).Interior.ColorIndex = xlNone
            ws.Cells(rowNum, colNum).Font.Color = RGB(156, 163, 175) ' Muted gray (#9CA3AF) for zero cells
        End If
    End If
End Sub

' Validate single sheet: checks if all row sums equal 1.000
Function ValidateSingleSheet(ws As Worksheet) As Boolean
    Dim matHeaderRow As Long, matStartCol As Long, colScan As Long, matEndCol As Long
    matStartCol = 3
    matHeaderRow = 6
    Dim rScanV As Long
    For rScanV = 4 To 8
        Dim hScanV As String: hScanV = Trim(CStr(ws.Cells(rScanV, 2).Value))
        If InStr(1, hScanV, "From", vbTextCompare) > 0 Or InStr(1, hScanV, "Dari", vbTextCompare) > 0 Then
            matHeaderRow = rScanV
            Exit For
        End If
    Next rScanV
    colScan = matStartCol
    Do While ws.Cells(matHeaderRow, colScan).Value <> "" And _
             ws.Cells(matHeaderRow, colScan).Value <> "Grand Total" And _
             ws.Cells(matHeaderRow, colScan).Value <> "Total Baris"
        colScan = colScan + 1
        If colScan > 200 Then Exit Do
    Loop
    matEndCol = colScan - 1
    Dim nClasses As Long: nClasses = matEndCol - matStartCol + 1
    If nClasses <= 0 Then
        ValidateSingleSheet = True
        Exit Function
    End If
    
    Dim r As Long, c As Long, rowSum As Double
    For r = matHeaderRow + 1 To matHeaderRow + nClasses
        rowSum = 0
        For c = matStartCol To matEndCol
            If IsNumeric(ws.Cells(r, c).Value) Then
                rowSum = rowSum + SafeCDbl(ws.Cells(r, c).Value)
            End If
        Next c
        If Abs(rowSum - 1#) > 0.0005 Then
            ValidateSingleSheet = False
            Exit Function
        End If
    Next r
    ValidateSingleSheet = True
End Function

' Count active interventions on a PU sheet (only user-intervened policies)
Function CountActiveInterventions(ws As Worksheet) As Long
    Dim matHeaderRow As Long, matStartCol As Long, colScan As Long
    matStartCol = 3
    matHeaderRow = 6
    Dim rScanC As Long
    For rScanC = 4 To 8
        Dim hScanC As String: hScanC = Trim(CStr(ws.Cells(rScanC, 2).Value))
        If InStr(1, hScanC, "From", vbTextCompare) > 0 Or InStr(1, hScanC, "Dari", vbTextCompare) > 0 Then
            matHeaderRow = rScanC
            Exit For
        End If
    Next rScanC
    colScan = matStartCol
    Do While ws.Cells(matHeaderRow, colScan).Value <> "" And _
             ws.Cells(matHeaderRow, colScan).Value <> "Grand Total" And _
             ws.Cells(matHeaderRow, colScan).Value <> "Total Baris"
        colScan = colScan + 1
        If colScan > 200 Then Exit Do
    Loop
    Dim sumCol As Long: sumCol = colScan
    Dim pCol As Long: pCol = sumCol + 2
    Dim pScanTitleC As String: pScanTitleC = CStr(ws.Cells(matHeaderRow, pCol).Value)
    If InStr(1, pScanTitleC, "From Class", vbTextCompare) = 0 And _
       InStr(1, pScanTitleC, "Kelas Asal", vbTextCompare) = 0 Then
        CountActiveInterventions = 0
        Exit Function
    End If
    
    Dim pStart As Long, pEnd As Long
    pStart = matHeaderRow + 1
    pEnd = ws.Cells(ws.Rows.Count, pCol).End(xlUp).Row
    
    Dim r As Long, count As Long: count = 0
    For r = pStart To pEnd
        Dim st As String: st = UCase(Trim(ws.Cells(r, pCol + 7).Value))
        If st = "INTERVENED" Or st = "DIINTERVENSI" Or st = "LOCKED" Then
            count = count + 1
        End If
    Next r
    CountActiveInterventions = count
End Function

' Update Dashboard Summary Table dynamically (Row 12 headers, Row 13+ data)
Sub UpdateDashboardSummary()
    Dim dashWs As Worksheet
    On Error Resume Next
    Set dashWs = ThisWorkbook.Sheets("Dashboard")
    On Error GoTo 0
    If dashWs Is Nothing Then Exit Sub
    
    Dim lastRow As Long
    lastRow = dashWs.Cells(dashWs.Rows.Count, 2).End(xlUp).Row
    Dim startSummaryRow As Long, scanR As Long
    startSummaryRow = 0
    For scanR = 10 To lastRow
        Dim hValS As String: hValS = Trim(CStr(dashWs.Cells(scanR, 2).Value))
        If hValS = "PU ID" Or hValS = "ID PU" Then
            startSummaryRow = scanR + 1
            Exit For
        End If
    Next scanR
    If startSummaryRow = 0 Then startSummaryRow = 18
    If lastRow < startSummaryRow Then Exit Sub
    
    Dim ws As Worksheet, r As Long
    For r = startSummaryRow To lastRow
        Dim puIdCell As Range
        Set puIdCell = dashWs.Cells(r, 2)
        If Trim(CStr(puIdCell.Value)) = "" Then Exit For
        
        Dim targetSheetName As String: targetSheetName = ""
        Dim puIdStr As String: puIdStr = Trim(CStr(puIdCell.Value))
        
        ' Look through sheets to find match
        For Each ws In ThisWorkbook.Worksheets
            If ws.Name <> "Dashboard" And Left(ws.Name, 1) <> "_" Then
                If InStr(1, ws.Name, "PU_" & puIdStr & "_") = 1 Or ws.Name = "PU_" & puIdStr Then
                    targetSheetName = ws.Name
                    Exit For
                End If
            End If
        Next ws
        
        If targetSheetName <> "" Then
            Set ws = ThisWorkbook.Sheets(targetSheetName)
            dashWs.Cells(r, 5).Value = CountActiveInterventions(ws)
            If ValidateSingleSheet(ws) Then
                dashWs.Cells(r, 6).Value = StatusValid()
                dashWs.Cells(r, 6).Interior.Color = RGB(210, 245, 210)
                dashWs.Cells(r, 6).Font.Color = RGB(0, 100, 0)
                dashWs.Cells(r, 6).Font.Bold = True
            Else
                dashWs.Cells(r, 6).Value = StatusWarning()
                dashWs.Cells(r, 6).Interior.Color = RGB(255, 210, 210)
                dashWs.Cells(r, 6).Font.Color = RGB(150, 0, 0)
                dashWs.Cells(r, 6).Font.Bold = True
            End If
        End If
    Next r
End Sub

' Validate All Sheets button macro
Sub ValidateAllSheets()
    UpdateDashboardSummary
    
    Dim dashWs As Worksheet
    Set dashWs = ThisWorkbook.Sheets("Dashboard")
    
    Dim lastRow As Long
    lastRow = dashWs.Cells(dashWs.Rows.Count, 2).End(xlUp).Row

    For scanR2 = 10 To lastRow

        End If

    validPUs = 0

        End If

        Else
            ShowMsg "Validation Warning: " & invalidCount & " of " & totalPUs & " Planning Units have row sum deviations! Please review the Dashboard.", vbExclamation, "SCIENDO Matrix Validation"
        End If
    End If
End Sub

' Reset specific sheet to baseline
Public Sub ResetCurrentSheetToBaseline(Optional ByVal wsTarget As Variant)
    Dim ws As Worksheet
    If IsMissing(wsTarget) Or IsEmpty(wsTarget) Then
        If Not ActiveSheet Is Nothing Then Set ws = ActiveSheet
    ElseIf VarType(wsTarget) = vbString Then
        If CStr(wsTarget) <> "" Then
            On Error Resume Next
            Set ws = ThisWorkbook.Sheets(CStr(wsTarget))
            On Error GoTo 0
        End If
    ElseIf TypeOf wsTarget Is Worksheet Then
        Set ws = wsTarget
    Else
        If Not ActiveSheet Is Nothing Then Set ws = ActiveSheet
    End If
    If ws Is Nothing Then Exit Sub
    If ws.Name = "Dashboard" Or Left(ws.Name, 1) = "_" Then Exit Sub
    
    Dim baseWs As Worksheet
    Set baseWs = GetBaseSheet(ws)
    
    If baseWs Is Nothing Then
        If IsIndonesian() Then
            ShowMsg "Lembar baseline untuk '" & ws.Name & "' tidak ditemukan.", vbExclamation, "Baseline Tidak Ditemukan"
        Else
            ShowMsg "Baseline sheet for '" & ws.Name & "' not found.", vbExclamation, "Baseline Not Found"
        End If
        Exit Sub
    End If
    
    IsUpdating = True
    Dim prevEvents As Boolean
    prevEvents = Application.EnableEvents
    Application.EnableEvents = False
    Application.ScreenUpdating = False
    
    On Error GoTo CleanExit
    
    ' Direct matrix coordinates
    Dim matHeaderRow As Long, matLabelCol As Long, matStartCol As Long, colScan As Long, matEndCol As Long
    matLabelCol = 2
    matStartCol = 3
    matHeaderRow = 6
    Dim rScanR As Long
    For rScanR = 4 To 8
        Dim hScanR As String: hScanR = Trim(CStr(ws.Cells(rScanR, matLabelCol).Value))
        If InStr(1, hScanR, "From", vbTextCompare) > 0 Or InStr(1, hScanR, "Dari", vbTextCompare) > 0 Then
            matHeaderRow = rScanR
            Exit For
        End If
    Next rScanR
    colScan = matStartCol
    Do While ws.Cells(matHeaderRow, colScan).Value <> "" And _
             ws.Cells(matHeaderRow, colScan).Value <> "Grand Total" And _
             ws.Cells(matHeaderRow, colScan).Value <> "Total Baris"
        colScan = colScan + 1
        If colScan > 200 Then Exit Do
    Loop
    matEndCol = colScan - 1
    Dim nClasses As Long: nClasses = matEndCol - matStartCol + 1
    Dim matEndRow As Long: matEndRow = matHeaderRow + nClasses
    Dim sumCol As Long: sumCol = matEndCol + 1
    
    ' Copy baseline matrix in one fast block assignment
    ws.Range(ws.Cells(matHeaderRow + 1, matStartCol), ws.Cells(matEndRow, matEndCol)).Value = _
        baseWs.Range(baseWs.Cells(matHeaderRow + 1, matStartCol), baseWs.Cells(matEndRow, matEndCol)).Value
    
    ' Restore cell formatting and update row sums
    Dim r As Long, c As Long
    For r = matHeaderRow + 1 To matEndRow
        For c = matStartCol To matEndCol
            RestoreCellFormatting ws, r, c, matHeaderRow, matLabelCol
        Next c
        UpdateRowSumOnSheet ws, r, matStartCol, matEndCol, sumCol
    Next r
    
    ' Reset panel
    Dim pCol As Long, pStart As Long, wsPEnd As Long, basePEnd As Long
    pCol = sumCol + 2
    Dim pScanTitleR As String: pScanTitleR = CStr(ws.Cells(matHeaderRow, pCol).Value)
    If InStr(1, pScanTitleR, "From Class", vbTextCompare) > 0 Or InStr(1, pScanTitleR, "Kelas Asal", vbTextCompare) > 0 Then
        pStart = matHeaderRow + 1
        wsPEnd = ws.Cells(ws.Rows.Count, pCol).End(xlUp).Row
        basePEnd = baseWs.Cells(baseWs.Rows.Count, pCol).End(xlUp).Row
        
        ' 1. Clear any dynamically appended rows beyond basePEnd
        If wsPEnd > basePEnd And wsPEnd >= pStart Then
            Dim cleanStart As Long
            cleanStart = IIf(basePEnd >= pStart, basePEnd + 1, pStart)
            With ws.Range(ws.Cells(cleanStart, pCol), ws.Cells(wsPEnd, pCol + 9))
                .ClearContents
                .Interior.ColorIndex = xlNone
                .Font.Bold = False
                .Borders.LineStyle = xlNone
            End With
        End If
        
        ' 2. Restore baseline rows
        If basePEnd >= pStart Then
            ws.Range(ws.Cells(pStart, pCol), ws.Cells(basePEnd, pCol + 9)).Value = _
                baseWs.Range(baseWs.Cells(pStart, pCol), baseWs.Cells(basePEnd, pCol + 9)).Value
            ' Re-apply Scenario Area dynamic formulas
            ws.Range(ws.Cells(pStart, pCol + 6), ws.Cells(basePEnd, pCol + 6)).FormulaR1C1 = "=ROUND(RC[-3]*(1+RC[-2]), 2)"
            ws.Range(ws.Cells(pStart, pCol + 7), ws.Cells(basePEnd, pCol + 7)).Font.Bold = False
            ws.Range(ws.Cells(pStart, pCol + 7), ws.Cells(basePEnd, pCol + 7)).Interior.ColorIndex = xlNone
        End If
    End If

CleanExit:
    Application.ScreenUpdating = True
    Application.EnableEvents = prevEvents
    IsUpdating = False
End Sub

' Reset All Sheets to Baseline button macro
Sub ResetAllSheetsToBaseline()
    If Application.Visible And Application.DisplayAlerts Then
        Dim confirmMsg As String, confirmTitle As String
        If IsIndonesian() Then
            confirmMsg = "Apakah Anda yakin ingin mengatur ulang semua Unit Perencanaan kembali ke laju transisi baseline historis? Semua penyesuaian skenario akan dihapus."
            confirmTitle = "Atur Ulang Semua ke Baseline"
        Else
            confirmMsg = "Are you sure you want to reset all Planning Units back to their baseline historical transition rates? All scenario adjustments will be cleared."
            confirmTitle = "Reset All to Baseline"
        End If
        If MsgBox(confirmMsg, vbQuestion + vbYesNo, confirmTitle) <> vbYes Then
            Exit Sub
        End If
    End If
    
    Dim prevEvents As Boolean
    prevEvents = Application.EnableEvents
    Application.EnableEvents = False
    Application.ScreenUpdating = False
    
    On Error GoTo ResetAllCleanExit
    
    Dim ws As Worksheet
    For Each ws In ThisWorkbook.Worksheets
        If ws.Name <> "Dashboard" And Left(ws.Name, 1) <> "_" Then

        End If

        ShowMsg "All Planning Units have been successfully reset to baseline.", vbInformation, "Reset Complete"
    End If
End Sub

' Helper function to apply policy to a single sheet
Function ApplyPolicyToSheet(ws As Worksheet, fromClass As String, toClass As String, pctVal As Double) As Boolean
    Dim matHeaderRow As Long, matLabelCol As Long, matStartCol As Long, colScan As Long
    matLabelCol = 2
    matStartCol = 3
    matHeaderRow = 6
    Dim rScanAP As Long
    For rScanAP = 4 To 8
        Dim hScanAP As String: hScanAP = Trim(CStr(ws.Cells(rScanAP, matLabelCol).Value))
        If InStr(1, hScanAP, "From", vbTextCompare) > 0 Or InStr(1, hScanAP, "Dari", vbTextCompare) > 0 Then
            matHeaderRow = rScanAP
            Exit For
        End If
    Next rScanAP
    colScan = matStartCol
    Do While ws.Cells(matHeaderRow, colScan).Value <> "" And _
             ws.Cells(matHeaderRow, colScan).Value <> "Grand Total" And _
             ws.Cells(matHeaderRow, colScan).Value <> "Total Baris"
        colScan = colScan + 1
        If colScan > 200 Then Exit Do
    Loop
    Dim matEndCol As Long: matEndCol = colScan - 1
    Dim nClasses As Long: nClasses = matEndCol - matStartCol + 1
    If nClasses <= 0 Then
        ApplyPolicyToSheet = False
        Exit Function
    End If
    
    Dim matEndRow As Long: matEndRow = matHeaderRow + nClasses
    Dim sumCol As Long: sumCol = colScan
    Dim pCol As Long: pCol = sumCol + 2
    Dim pScanTitleAP As String: pScanTitleAP = CStr(ws.Cells(matHeaderRow, pCol).Value)
    If InStr(1, pScanTitleAP, "From Class", vbTextCompare) = 0 And _
       InStr(1, pScanTitleAP, "Kelas Asal", vbTextCompare) = 0 Then
        ApplyPolicyToSheet = False
        Exit Function
    End If
    
    Dim pStart As Long, pEnd As Long
    pStart = matHeaderRow + 1
    pEnd = ws.Cells(ws.Rows.Count, pCol).End(xlUp).Row
    
    ' Find matrix row for fromClass
    Dim mRow As Long, rIdx As Long
    mRow = 0
    For rIdx = matHeaderRow + 1 To matEndRow
        If StrComp(Trim(CStr(ws.Cells(rIdx, matLabelCol).Value)), fromClass, vbTextCompare) = 0 Then
            mRow = rIdx
            Exit For
        End If
    Next rIdx
    If mRow = 0 Then
        For rIdx = matHeaderRow + 1 To matEndRow
            If InStr(1, Trim(CStr(ws.Cells(rIdx, matLabelCol).Value)), fromClass, vbTextCompare) > 0 Then
                mRow = rIdx
                Exit For
            End If
        Next rIdx
    End If
    If mRow = 0 Then
        ApplyPolicyToSheet = False
        Exit Function
    End If
    
    ' Find matrix col for toClass
    Dim mCol As Long, cIdx As Long
    mCol = 0
    For cIdx = matStartCol To matEndCol
        If StrComp(Trim(CStr(ws.Cells(matHeaderRow, cIdx).Value)), toClass, vbTextCompare) = 0 Then
            mCol = cIdx
            Exit For
        End If
    Next cIdx
    If mCol = 0 Then
        For cIdx = matStartCol To matEndCol
            If InStr(1, Trim(CStr(ws.Cells(matHeaderRow, cIdx).Value)), toClass, vbTextCompare) > 0 Then
                mCol = cIdx
                Exit For
            End If
        Next cIdx
    End If
    If mCol = 0 Then
        ApplyPolicyToSheet = False
        Exit Function
    End If
    
    Dim baseWs As Worksheet
    Set baseWs = GetBaseSheet(ws)
    
    ' CASE A: fromClass == toClass (Diagonal persistence)
    If (mRow - matHeaderRow) = (mCol - matLabelCol) Then
        IsUpdating = True
        Dim prevEventsDiag As Boolean: prevEventsDiag = Application.EnableEvents
        Application.EnableEvents = False
        
        If Abs(pctVal - 1#) < 0.0001 Or pctVal >= 1# Then
            ' 100% persistence moratorium
            ws.Cells(mRow, mCol).Value = 1#
            ws.Cells(mRow, mCol).Interior.Color = RGB(255, 235, 130)
            
            ' Rebalance all other cells in row to 0
            Dim c As Long
            For c = matStartCol To matEndCol
                If c <> mCol Then
                    ws.Cells(mRow, c).Value = 0#
                    RestoreCellFormatting ws, mRow, c, matHeaderRow, matLabelCol
                End If
            Next c
            
            ' Update all panel rows for mRow
            Dim pr As Long
            pr = pStart
            Do While pr <= pEnd And pEnd >= pStart
                If Val(ws.Cells(pr, pCol + 8).Value) = mRow Then
                    Dim bRateD As Double
                    bRateD = SafeCDbl(ws.Cells(pr, pCol + 2).Value)
                    If bRateD > 0.000001 Then
                        ws.Cells(pr, pCol + 4).Value = -1#
                        ws.Cells(pr, pCol + 5).Value = 0#
                        ws.Cells(pr, pCol + 6).FormulaR1C1 = "=ROUND(RC[-3]*(1+RC[-2]), 2)"
                        ws.Cells(pr, pCol + 7).Value = StatusIntervened()
                        ws.Cells(pr, pCol + 7).Font.Bold = True
                        ws.Cells(pr, pCol + 7).Interior.Color = RGB(255, 235, 130)
                        pr = pr + 1
                    Else
                        ' Delete previously appended 0-to-exist row since rate is now 0
                        DeletePanelRow ws, pr, pCol
                        pEnd = ws.Cells(ws.Rows.Count, pCol).End(xlUp).Row
                    End If
                Else
                    pr = pr + 1
                End If
            Loop
            
            UpdateRowSumOnSheet ws, mRow, matStartCol, matEndCol, sumCol
        Else
            ' Arbitrary persistence adjustment (e.g. +10% or -20%)
            Dim baseDiagRate As Double
            If Not baseWs Is Nothing Then
                baseDiagRate = SafeCDbl(baseWs.Cells(mRow, mCol).Value)
            Else
                baseDiagRate = SafeCDbl(ws.Cells(mRow, mCol).Value)
            End If
            
            Dim newDiagRate As Double
            newDiagRate = baseDiagRate * (1# + pctVal)
            If newDiagRate < 0# Then newDiagRate = 0#
            If newDiagRate > 1# Then newDiagRate = 1#
            
            ws.Cells(mRow, mCol).Value = newDiagRate
            ws.Cells(mRow, mCol).Interior.Color = RGB(255, 235, 130)
            
            Dim remP As Double: remP = 1# - newDiagRate
            Dim sumOffDiagBase As Double: sumOffDiagBase = 0#
            For c = matStartCol To matEndCol
                If c <> mCol Then
                    If Not baseWs Is Nothing Then
                        sumOffDiagBase = sumOffDiagBase + SafeCDbl(baseWs.Cells(mRow, c).Value)
                    Else
                        sumOffDiagBase = sumOffDiagBase + SafeCDbl(ws.Cells(mRow, c).Value)
                    End If
                End If
            Next c
            
            If remP <= 0.000001 Then
                For c = matStartCol To matEndCol
                    If c <> mCol Then
                        ws.Cells(mRow, c).Value = 0#
                        RestoreCellFormatting ws, mRow, c, matHeaderRow, matLabelCol
                    End If
                Next c
                pr = pStart
                Do While pr <= pEnd And pEnd >= pStart
                    If Val(ws.Cells(pr, pCol + 8).Value) = mRow Then
                        Dim bRateZ As Double: bRateZ = SafeCDbl(ws.Cells(pr, pCol + 2).Value)
                        If bRateZ > 0.000001 Then
                            ws.Cells(pr, pCol + 4).Value = -1#
                            ws.Cells(pr, pCol + 5).Value = 0#
                            ws.Cells(pr, pCol + 6).FormulaR1C1 = "=ROUND(RC[-3]*(1+RC[-2]), 2)"
                            ws.Cells(pr, pCol + 7).Value = StatusIntervened()
                            ws.Cells(pr, pCol + 7).Font.Bold = True
                            ws.Cells(pr, pCol + 7).Interior.Color = RGB(255, 235, 130)
                            pr = pr + 1
                        Else
                            DeletePanelRow ws, pr, pCol
                            pEnd = ws.Cells(ws.Rows.Count, pCol).End(xlUp).Row
                        End If
                    Else
                        pr = pr + 1
                    End If
                Loop
            ElseIf sumOffDiagBase > 0.000001 Then
                For c = matStartCol To matEndCol
                    If c <> mCol Then
                        Dim bCVal As Double
                        If Not baseWs Is Nothing Then bCVal = SafeCDbl(baseWs.Cells(mRow, c).Value) Else bCVal = SafeCDbl(ws.Cells(mRow, c).Value)
                        Dim cRate As Double
                        cRate = (bCVal / sumOffDiagBase) * remP
                        ws.Cells(mRow, c).Value = cRate
                        RestoreCellFormatting ws, mRow, c, matHeaderRow, matLabelCol
                    End If
                Next c
                pr = pStart
                Do While pr <= pEnd And pEnd >= pStart
                    If Val(ws.Cells(pr, pCol + 8).Value) = mRow Then
                        Dim targetCol As Long: targetCol = Val(ws.Cells(pr, pCol + 9).Value)
                        Dim rateVal As Double: rateVal = SafeCDbl(ws.Cells(mRow, targetCol).Value)
                        Dim bRateVal As Double: bRateVal = SafeCDbl(ws.Cells(pr, pCol + 2).Value)
                        
                        If bRateVal > 0.000001 Then
                            ws.Cells(pr, pCol + 5).Value = rateVal
                            ws.Cells(pr, pCol + 4).Value = (rateVal - bRateVal) / bRateVal
                            ws.Cells(pr, pCol + 6).FormulaR1C1 = "=ROUND(RC[-3]*(1+RC[-2]), 2)"
                            ws.Cells(pr, pCol + 7).Value = StatusIntervened()
                            ws.Cells(pr, pCol + 7).Font.Bold = True
                            ws.Cells(pr, pCol + 7).Interior.Color = RGB(255, 235, 130)
                            pr = pr + 1
                        Else
                            ' Appended 0-to-exist row
                            If rateVal <= 0.000001 Then
                                DeletePanelRow ws, pr, pCol
                                pEnd = ws.Cells(ws.Rows.Count, pCol).End(xlUp).Row
                            Else
                                ws.Cells(pr, pCol + 5).Value = rateVal
                                ws.Cells(pr, pCol + 4).NumberFormat = "0.0%"
                                ws.Cells(pr, pCol + 4).Value = rateVal
                                Dim initADiag As Double
                                initADiag = GetInitialClassArea(ws, CStr(ws.Cells(pr, pCol).Value), mRow, pCol, pStart, pEnd)
                                ws.Cells(pr, pCol + 6).FormulaR1C1 = "=ROUND(" & Replace(CStr(initADiag), ",", ".") & "*RC[-1], 2)"
                                ws.Cells(pr, pCol + 7).Value = StatusIntervened()
                                ws.Cells(pr, pCol + 7).Font.Bold = True
                                ws.Cells(pr, pCol + 7).Interior.Color = RGB(255, 235, 130)
                                pr = pr + 1
                            End If
                        End If
                    Else
                        pr = pr + 1
                    End If
                Loop
            Else
                ' sumOffDiagBase is 0 (100% persistence in baseline) and remP > 0
                Dim nOffDiag As Long: nOffDiag = (matEndCol - matStartCol)
                If nOffDiag > 0 Then
                    Dim eqShare As Double: eqShare = remP / nOffDiag
                    Dim runShare As Double: runShare = 0#
                    Dim lastOffCol As Long: lastOffCol = 0
                    For c = matStartCol To matEndCol
                        If c <> mCol Then lastOffCol = c
                    Next c
                    For c = matStartCol To matEndCol
                        If c <> mCol Then
                            Dim cellValC As Double
                            If c = lastOffCol Then
                                cellValC = remP - runShare
                                If cellValC < 0# Then cellValC = 0#
                            Else
                                cellValC = eqShare
                                runShare = runShare + eqShare
                            End If
                            ws.Cells(mRow, c).Value = cellValC
                            RestoreCellFormatting ws, mRow, c, matHeaderRow, matLabelCol
                        End If
                    Next c
                End If
            End If
            UpdateRowSumOnSheet ws, mRow, matStartCol, matEndCol, sumCol
        End If
        
        Application.EnableEvents = prevEventsDiag
        IsUpdating = False
        ApplyPolicyToSheet = True
        Exit Function
    End If
    
    ' CASE B: fromClass != toClass (Off-diagonal transition)
    Dim pRow As Long
    pRow = FindPanelRow(ws, mRow, mCol, pCol, pStart, pEnd)
    
    If pRow > 0 Then
        ' Transition already exists in panel
        Dim baseRate As Double, scenRate As Double, effPct As Double
        baseRate = SafeCDbl(ws.Cells(pRow, pCol + 2).Value)
        
        If baseRate > 0.000001 Then
            scenRate = baseRate * (1# + pctVal)
            If scenRate < 0 Then scenRate = 0
            If scenRate > 1 Then scenRate = 1
            effPct = (scenRate - baseRate) / baseRate
            
            If ValidateLockedSum(ws, mRow, mCol, scenRate, matStartCol, matEndCol, pCol, pStart, pEnd) Then
                IsUpdating = True
                Dim prevEventsB1 As Boolean: prevEventsB1 = Application.EnableEvents
                Application.EnableEvents = False
                
                ws.Cells(pRow, pCol + 4).Value = effPct
                ws.Cells(pRow, pCol + 5).Value = scenRate
                ws.Cells(pRow, pCol + 6).FormulaR1C1 = "=ROUND(RC[-3]*(1+RC[-2]), 2)"
                ws.Cells(pRow, pCol + 7).Value = StatusIntervened()
                ws.Cells(pRow, pCol + 7).Font.Bold = True
                ws.Cells(pRow, pCol + 7).Interior.Color = RGB(255, 235, 130)
                ws.Cells(mRow, mCol).Value = scenRate
                ws.Cells(mRow, mCol).Interior.Color = RGB(255, 235, 130)
                
                RebalanceRow ws, mRow, mCol, scenRate, matStartCol, matEndCol, pCol, pStart, pEnd
                UpdateRowSumOnSheet ws, mRow, matStartCol, matEndCol, sumCol
                
                Application.EnableEvents = prevEventsB1
                IsUpdating = False
                ApplyPolicyToSheet = True
                Exit Function
            End If
        Else
            ' Appended row already existed
            scenRate = pctVal
            If scenRate < 0 Then scenRate = 0
            If scenRate > 1 Then scenRate = 1
            If scenRate <= 0.000001 Then
                ' Set to 0 -> delete appended row
                IsUpdating = True
                Dim prevEventsB2 As Boolean: prevEventsB2 = Application.EnableEvents
                Application.EnableEvents = False
                
                ws.Cells(mRow, mCol).Value = 0#
                RestoreCellFormatting ws, mRow, mCol, matHeaderRow, matLabelCol
                DeletePanelRow ws, pRow, pCol
                pEnd = ws.Cells(ws.Rows.Count, pCol).End(xlUp).Row
                RebalanceRow ws, mRow, mCol, 0#, matStartCol, matEndCol, pCol, pStart, pEnd
                UpdateRowSumOnSheet ws, mRow, matStartCol, matEndCol, sumCol
                
                Application.EnableEvents = prevEventsB2
                IsUpdating = False
                ApplyPolicyToSheet = True
                Exit Function
            Else
                If ValidateLockedSum(ws, mRow, mCol, scenRate, matStartCol, matEndCol, pCol, pStart, pEnd) Then
                    IsUpdating = True
                    Dim prevEventsB3 As Boolean: prevEventsB3 = Application.EnableEvents
                    Application.EnableEvents = False
                    
                    ws.Cells(pRow, pCol + 4).NumberFormat = "0.0%"
                    ws.Cells(pRow, pCol + 4).Value = scenRate
                    ws.Cells(pRow, pCol + 5).Value = scenRate
                    Dim iArea1 As Double
                    iArea1 = GetInitialClassArea(ws, CStr(ws.Cells(mRow, matLabelCol).Value), mRow, pCol, pStart, pEnd)
                    ws.Cells(pRow, pCol + 6).FormulaR1C1 = "=ROUND(" & Replace(CStr(iArea1), ",", ".") & "*RC[-1], 2)"
                    ws.Cells(pRow, pCol + 7).Value = StatusIntervened()
                    ws.Cells(pRow, pCol + 7).Font.Bold = True
                    ws.Cells(pRow, pCol + 7).Interior.Color = RGB(255, 235, 130)
                    ws.Cells(mRow, mCol).Value = scenRate
                    ws.Cells(mRow, mCol).Interior.Color = RGB(255, 235, 130)
                    
                    RebalanceRow ws, mRow, mCol, scenRate, matStartCol, matEndCol, pCol, pStart, pEnd
                    UpdateRowSumOnSheet ws, mRow, matStartCol, matEndCol, sumCol
                    
                    Application.EnableEvents = prevEventsB3
                    IsUpdating = False
                    ApplyPolicyToSheet = True
                    Exit Function
                End If
            End If
        End If
    Else
        ' Transition does NOT exist in panel (0-to-exist intervention via Batch Policy!)
        If pctVal > 0.000001 Then
            Dim newRate As Double: newRate = pctVal
            If newRate > 1# Then newRate = 1#
            
            If ValidateLockedSum(ws, mRow, mCol, newRate, matStartCol, matEndCol, pCol, pStart, pEnd) Then
                IsUpdating = True
                Dim prevEventsB4 As Boolean: prevEventsB4 = Application.EnableEvents
                Application.EnableEvents = False
                
                AppendPanelRow ws, mRow, mCol, newRate, matHeaderRow, matLabelCol, pCol, pStart, pEnd
                
                ws.Cells(mRow, mCol).Value = newRate
                ws.Cells(mRow, mCol).Interior.Color = RGB(255, 235, 130)
                
                RebalanceRow ws, mRow, mCol, newRate, matStartCol, matEndCol, pCol, pStart, pEnd
                UpdateRowSumOnSheet ws, mRow, matStartCol, matEndCol, sumCol
                
                Application.EnableEvents = prevEventsB4
                IsUpdating = False
                ApplyPolicyToSheet = True
                Exit Function
            End If
        End If
    End If
    
    ApplyPolicyToSheet = False
End Function

' Apply Policy Across PUs button macro
' Reads From Class (C8), To Class (C9), % Adjustment (E8), and Target PU (E9) from Dashboard
Sub ApplyPolicyAcrossPUs()
    Dim dashWs As Worksheet
    On Error Resume Next
    Set dashWs = ThisWorkbook.Sheets("Dashboard")
    On Error GoTo 0
    If dashWs Is Nothing Then Exit Sub
    
    Dim fromClass As String, toClass As String, targetPU As String
    fromClass = Trim(CStr(dashWs.Range("C8").Value))
    toClass   = Trim(CStr(dashWs.Range("C9").Value))
    
    If fromClass = "" Or toClass = "" Then
        If IsIndonesian() Then
            ShowMsg "Silakan pilih 'Kelas Asal' dan 'Kelas Tujuan' pada Dasbor sebelum menerapkan kebijakan.", vbExclamation, "Pemilihan Kelas Belum Lengkap"
        Else
            ShowMsg "Please select both 'From Class' and 'To Class' on the Dashboard before applying policy.", vbExclamation, "Missing Class Selection"
        End If
        Exit Sub
    End If
    
    Dim pctRaw As Variant
    pctRaw = dashWs.Range("E8").Value
    If Not IsNumeric(pctRaw) Then
        If IsIndonesian() Then
            ShowMsg "Silakan masukkan persentase penyesuaian yang valid pada sel E8 (contoh: -50%, 100%, atau 0.05).", vbExclamation, "Persentase Tidak Valid"
        Else
            ShowMsg "Please enter a valid percentage adjustment in cell E8 (e.g., -50%, 100%, or 0.05).", vbExclamation, "Invalid Percentage"
        End If
        Exit Sub
    End If
    Dim pctVal As Double
    pctVal = SafeCDbl(pctRaw)
    If Abs(pctVal) > 1# Then
        pctVal = pctVal / 100#
    End If
    
    targetPU = Trim(CStr(dashWs.Range("E9").Value))
    If targetPU = "" Then targetPU = "ALL"
    
    Dim isAll As Boolean
    isAll = (UCase(targetPU) = "ALL" Or InStr(1, targetPU, "All Planning Units", vbTextCompare) > 0 Or InStr(1, targetPU, "Semua Unit Perencanaan", vbTextCompare) > 0)
    
    ' Extract PU ID if specific PU chosen (e.g. "PU 1: ...", "PU 1", or numeric "1")
    Dim targetId As String: targetId = ""
    If Not isAll Then
        Dim matchId As String: matchId = ""
        Dim chIdx As Long
        For chIdx = 1 To Len(targetPU)
            Dim ch As String: ch = Mid(targetPU, chIdx, 1)
            If ch >= "0" And ch <= "9" Then
                matchId = matchId & ch
            ElseIf Len(matchId) > 0 Then
                Exit For
            End If
        Next chIdx
        targetId = matchId
    End If
    
    Dim ws As Worksheet, modifiedCount As Long: modifiedCount = 0
    For Each ws In ThisWorkbook.Worksheets
        If ws.Name <> "Dashboard" And Left(ws.Name, 1) <> "_" Then
            Dim shouldApply As Boolean: shouldApply = False
            If isAll Then
                shouldApply = True
            ElseIf targetId <> "" Then
                If InStr(1, ws.Name, "PU_" & targetId & "_") = 1 Or ws.Name = "PU_" & targetId Then
                    shouldApply = True
                End If
            Else
                If InStr(1, ws.Name, targetPU, vbTextCompare) > 0 Then
                    shouldApply = True
                End If
            End If
            
            If shouldApply Then
                If ApplyPolicyToSheet(ws, fromClass, toClass, pctVal) Then
                    modifiedCount = modifiedCount + 1
                End If
            End If
        End If
    Next ws
    
    UpdateDashboardSummary
    If IsIndonesian() Then
        ShowMsg "Kebijakan berhasil diterapkan pada " & modifiedCount & " Unit Perencanaan." & vbCrLf & _
                "Transisi: " & fromClass & " -> " & toClass & vbCrLf & _
                "Penyesuaian: " & Format(pctVal, "0.0%"), vbInformation, "Penerapan Kebijakan Selesai"
    Else
        ShowMsg "Policy applied to " & modifiedCount & " Planning Unit(s)." & vbCrLf & _
                "Transition: " & fromClass & " -> " & toClass & vbCrLf & _
                "Adjustment: " & Format(pctVal, "0.0%"), vbInformation, "Policy Application Complete"
    End If
End Sub
